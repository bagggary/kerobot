cmake_minimum_required(VERSION 3.24)
project(KERobot VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
add_library(kerobot_core libs/geometry/src/core.cpp)
target_include_directories(kerobot_core PUBLIC libs/geometry/include)
if(MSVC)
  target_compile_options(kerobot_core PRIVATE /W4 /WX)
else()
  target_compile_options(kerobot_core PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()
add_executable(kerobot_offline apps/offline/main.cpp)
target_link_libraries(kerobot_offline PRIVATE kerobot_core)
include(CTest)
if(BUILD_TESTING)
  add_executable(kerobot_tests tests/unit/core_test.cpp)
  target_link_libraries(kerobot_tests PRIVATE kerobot_core)
  add_test(NAME offline_contracts_and_geometry COMMAND kerobot_tests)
  add_test(NAME offline_demo COMMAND kerobot_offline ${CMAKE_CURRENT_BINARY_DIR}/demo-job)
endif()
option(KEROBOT_BUILD_DESKTOP "Build the optional Qt review shell" OFF)
if(KEROBOT_BUILD_DESKTOP)
  find_package(Qt6 REQUIRED COMPONENTS Widgets)
  add_executable(kerobot_desktop apps/desktop/main.cpp)
  target_link_libraries(kerobot_desktop PRIVATE kerobot_core Qt6::Widgets)
endif()

option(KEROBOT_BUILD_STEP_PROBE "Build the standalone offline STEP inspection tool" OFF)
if(KEROBOT_BUILD_STEP_PROBE)
  find_package(Qt6 REQUIRED COMPONENTS Core)
  find_package(OpenCASCADE 7.8.1 EXACT CONFIG REQUIRED COMPONENTS
    FoundationClasses ModelingData ModelingAlgorithms Visualization ApplicationFramework DataExchange)
  add_executable(kerobot_step_probe workers/cad/step_probe.cpp)
  target_include_directories(kerobot_step_probe SYSTEM PRIVATE ${OpenCASCADE_INCLUDE_DIR})
  target_compile_definitions(kerobot_step_probe PRIVATE QT_NO_KEYWORDS)
  target_link_libraries(kerobot_step_probe PRIVATE Qt6::Core TKDESTEP TKXCAF TKTopAlgo)
  if(BUILD_TESTING)
    find_package(Python3 REQUIRED COMPONENTS Interpreter)
    add_test(NAME step_diagnostics COMMAND ${Python3_EXECUTABLE}
      ${CMAKE_CURRENT_SOURCE_DIR}/tests/golden/test_step.py $<TARGET_FILE:kerobot_step_probe>)
    set_tests_properties(step_diagnostics PROPERTIES TIMEOUT 60)
  endif()
endif()
